# 7.3  Human-in-the-Loop Implementation
# ---------------------------------------------------------------------------

class HITLState(TypedDict, total=False):
    spec: str
    topology: str
    human_approved: bool
    netlist: str
    status: str


def hitl_parse_spec(state: HITLState) -> dict:
    return {"spec": state.get("spec", ""), "status": "spec_parsed"}


def hitl_select_topology(state: HITLState) -> dict:
    return {"topology": "folded-cascode-OTA", "status": "topology_selected"}


def hitl_generate_netlist(state: HITLState) -> dict:
    approved = state.get("human_approved", False)
    if not approved:
        return {"status": "rejected_by_human", "netlist": ""}
    return {
        "netlist": f"* {state.get('topology', 'unknown')} netlist\n.end",
        "status": "netlist_generated",
    }


hitl_builder = StateGraph(HITLState)

hitl_builder.add_node("parse_spec", hitl_parse_spec)
hitl_builder.add_node("select_topology", hitl_select_topology)
hitl_builder.add_node("generate_netlist", hitl_generate_netlist)

hitl_builder.add_edge(START, "parse_spec")
hitl_builder.add_edge("parse_spec", "select_topology")
hitl_builder.add_edge("select_topology", "generate_netlist")
hitl_builder.add_edge("generate_netlist", END)

hitl_memory = MemorySaver()
hitl_graph = hitl_builder.compile(
    checkpointer=hitl_memory,
    interrupt_after=["select_topology"],
)

print("HITL graph compiled with interrupt_after=['select_topology']")
print("Execution will pause after topology is selected, awaiting human approval.\n")

# --- Phase 1: Run until interrupt ---
hitl_config = {"configurable": {"thread_id": "hitl-demo-001"}}
hitl_input = {"spec": "Design a low-noise OTA for biomedical ADC front-end"}

partial_state = hitl_graph.invoke(hitl_input, config=hitl_config)

print("Phase 1 - Execution paused after topology selection:")
print(f"  topology: {partial_state.get('topology')}")
print(f"  status  : {partial_state.get('status')}")
print(f"  netlist : {partial_state.get('netlist', '(not yet generated)')}")
print()

# --- Phase 2: Human approves and resumes ---
print(">>> HUMAN DECISION: Approved topology 'folded-cascode-OTA' <<<\n")

hitl_graph.update_state(hitl_config, {"human_approved": True})
final_hitl = hitl_graph.invoke(None, config=hitl_config)

print("Phase 2 - Execution resumed after human approval:")
print(f"  status  : {final_hitl.get('status')}")
print(f"  netlist : {final_hitl.get('netlist', '(empty)')[:60]}...")
