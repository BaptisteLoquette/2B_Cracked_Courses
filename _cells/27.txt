# ---------------------------------------------------------------------------
# 6.3  Time-Travel Implementation
# ---------------------------------------------------------------------------

memory_tt = MemorySaver()
eda_graph_tt = eda_builder.compile(checkpointer=memory_tt)

config_tt = {"configurable": {"thread_id": "time-travel-demo"}}

random.seed(123)
final_tt = eda_graph_tt.invoke(initial_state, config=config_tt)

print(f"Final status: {final_tt['status']}, iterations: {final_tt['iteration']}\n")

# Retrieve all checkpoints in reverse chronological order
all_checkpoints = list(memory_tt.list(config_tt))
print(f"Total checkpoints: {len(all_checkpoints)}\n")

# Walk through checkpoints and extract key metrics at each step
print(f"{'Step':>4} | {'Node':>20} | {'Iter':>4} | {'Status':>10} | {'Gain(dB)':>9} | {'PM(deg)':>8}")
print("-" * 70)

for cp in reversed(all_checkpoints):
    vals = cp.checkpoint.get("channel_values", {})
    if not vals:
        continue
    node_name = cp.metadata.get("source", "?")
    iteration = vals.get("iteration", "-")
    status = vals.get("status", "-")
    sim = vals.get("simulation_results", {})
    gain = sim.get("gain_db", "-")
    pm = sim.get("phase_margin_deg", "-")
    step = cp.metadata.get("step", "?")

    gain_str = f"{gain:>9.2f}" if isinstance(gain, (int, float)) else f"{gain:>9}"
    pm_str = f"{pm:>8.1f}" if isinstance(pm, (int, float)) else f"{pm:>8}"

    print(f"{step:>4} | {node_name:>20} | {str(iteration):>4} | {status:>10} | {gain_str} | {pm_str}")
