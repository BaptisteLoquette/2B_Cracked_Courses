import matplotlib.pyplot as plt
import matplotlib.patches as mpatches
import networkx as nx

# ---------------------------------------------------------------------------
# 2.4  Visualize the simple graph
# ---------------------------------------------------------------------------

def draw_langgraph(compiled_graph, title: str = "LangGraph Workflow") -> None:
    """Render a LangGraph compiled graph as a networkx diagram."""
    G = nx.DiGraph()

    graph_data = compiled_graph.get_graph()
    for node in graph_data.nodes:
        G.add_node(node)
    for edge in graph_data.edges:
        G.add_edge(edge.source, edge.target,
                   conditional=edge.conditional if hasattr(edge, 'conditional') else False)

    fig, ax = plt.subplots(1, 1, figsize=(10, 4))

    pos = nx.spring_layout(G, seed=42, k=2.5)

    node_colors = []
    for n in G.nodes():
        if n == "__start__":
            node_colors.append("#2ecc71")
        elif n == "__end__":
            node_colors.append("#e74c3c")
        else:
            node_colors.append("#3498db")

    nx.draw_networkx_nodes(G, pos, ax=ax, node_size=2200,
                           node_color=node_colors, edgecolors="black", linewidths=1.5)
    nx.draw_networkx_labels(G, pos, ax=ax, font_size=9, font_weight="bold",
                            font_color="white")

    solid_edges = [(u, v) for u, v, d in G.edges(data=True) if not d.get("conditional")]
    cond_edges = [(u, v) for u, v, d in G.edges(data=True) if d.get("conditional")]

    nx.draw_networkx_edges(G, pos, edgelist=solid_edges, ax=ax,
                           edge_color="#2c3e50", width=2, arrows=True,
                           arrowsize=20, arrowstyle="-|>")
    nx.draw_networkx_edges(G, pos, edgelist=cond_edges, ax=ax,
                           edge_color="#e67e22", width=2, style="dashed",
                           arrows=True, arrowsize=20, arrowstyle="-|>")

    legend_elements = [
        mpatches.Patch(facecolor="#2ecc71", edgecolor="black", label="START"),
        mpatches.Patch(facecolor="#e74c3c", edgecolor="black", label="END"),
        mpatches.Patch(facecolor="#3498db", edgecolor="black", label="Node"),
        plt.Line2D([0], [0], color="#2c3e50", lw=2, label="Edge"),
        plt.Line2D([0], [0], color="#e67e22", lw=2, ls="--", label="Conditional"),
    ]
    ax.legend(handles=legend_elements, loc="upper left", fontsize=8)
    ax.set_title(title, fontsize=13, fontweight="bold")
    ax.axis("off")
    plt.tight_layout()
    plt.show()
