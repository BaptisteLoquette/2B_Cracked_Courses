def parse_spec(state: EDAState) -> dict:
    """Parse the natural-language specification into structured targets."""
    spec_text = state.get("spec", "")
    targets = {
        "gain_db": 60.0,
        "ugf_mhz": 100.0,
        "phase_margin_deg": 60.0,
        "power_mw": 1.0,
        "area_um2": 2500.0,
        "supply_v": 1.8,
        "load_pf": 2.0,
    }
    return {
        "ppa_metrics": {"targets": targets},
        "iteration": 0,
        "max_iterations": state.get("max_iterations", 10),
        "status": "running",
        "error_log": [f"[parse_spec] Parsed spec: '{spec_text[:60]}...'"],
    }


def select_topology(state: EDAState) -> dict:
    """Select circuit topology based on specification targets."""
    targets = state.get("ppa_metrics", {}).get("targets", {})
    gain = targets.get("gain_db", 40)

    if gain > 50:
        topology = "folded-cascode-OTA"
        rationale = f"Gain target {gain} dB > 50 dB -> folded-cascode for high output impedance"
    elif gain > 30:
        topology = "telescopic-OTA"
        rationale = f"Gain target {gain} dB in [30,50] -> telescopic for moderate gain, low power"
    else:
        topology = "simple-diff-pair"
        rationale = f"Gain target {gain} dB <= 30 -> simple differential pair sufficient"

    errors = list(state.get("error_log", []))
    errors.append(f"[select_topology] Selected {topology}: {rationale}")
    return {"topology": topology, "error_log": errors}

def run_simulation(state: EDAState) -> dict:
    """Mock SPICE simulation returning realistic analog metrics."""
    iteration = state.get("iteration", 0)
    sizing = state.get("sizing_params", {})
    w = sizing.get("W_um", 10.0)
    ib = sizing.get("Ib_uA", 50.0)

    gm = 2 * ib * 1e-6 / 0.2
    ro = 50e3 / (1 + iteration * 0.1)
    gain_linear = gm * ro
    gain_db = 20 * math.log10(max(gain_linear, 1))

    ugf = (gm / (2 * math.pi * 2e-12)) / 1e6
    phase_margin = 45 + iteration * 3.5 + random.uniform(-2, 2)
    power_mw = 1.8 * ib * 1e-3 * 2
    area_um2 = w * 0.18 * 6 * 2

    sim_pass = random.random() > 0.1

    results = {
        "gain_db": round(gain_db, 2),
        "ugf_mhz": round(ugf, 2),
        "phase_margin_deg": round(min(phase_margin, 89), 2),
        "power_mw": round(power_mw, 4),
        "area_um2": round(area_um2, 2),
        "sim_converged": sim_pass,
    }
    errors = list(state.get("error_log", []))
    errors.append(f"[run_simulation] gain={results['gain_db']}dB, PM={results['phase_margin_deg']}deg, "
                  f"converged={sim_pass}")
    return {"simulation_results": results, "error_log": errors}


def evaluate_ppa(state: EDAState) -> dict:
    """Evaluate whether PPA targets are met."""
    targets = state.get("ppa_metrics", {}).get("targets", {})
    results = state.get("simulation_results", {})
    iteration = state.get("iteration", 0)

    if not results.get("sim_converged", False):
        errors = list(state.get("error_log", []))
        errors.append(f"[evaluate_ppa] Simulation did not converge - will re-size")
        return {
            "status": "running",
            "iteration": iteration + 1,
            "ppa_metrics": {**state.get("ppa_metrics", {}), "achieved": results},
            "error_log": errors,
        }

    met = {
        "gain": results.get("gain_db", 0) >= targets.get("gain_db", 0),
        "ugf": results.get("ugf_mhz", 0) >= targets.get("ugf_mhz", 0),
        "phase_margin": results.get("phase_margin_deg", 0) >= targets.get("phase_margin_deg", 0),
        "power": results.get("power_mw", 999) <= targets.get("power_mw", 999),
    }
    all_met = all(met.values())
    over_budget = iteration + 1 >= state.get("max_iterations", 10)

    if all_met:
        status = "converged"
    elif over_budget:
        status = "failed"
    else:
        status = "running"

    errors = list(state.get("error_log", []))
    errors.append(f"[evaluate_ppa] Targets met: {met}, status={status}")
    return {
        "status": status,
        "iteration": iteration + 1,
        "ppa_metrics": {**state.get("ppa_metrics", {}), "achieved": results, "met": met},
        "error_log": errors,
    }
