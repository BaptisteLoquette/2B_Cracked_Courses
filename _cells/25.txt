# ---------------------------------------------------------------------------
# 5.3  Visualize the routing decision tree
# ---------------------------------------------------------------------------

fig, ax = plt.subplots(figsize=(12, 7))

G_route = nx.DiGraph()
nodes_route = [
    ("evaluate_ppa", {"color": "#3498db"}),
    ("PM < 45?", {"color": "#f39c12"}),
    ("Gain < target?", {"color": "#f39c12"}),
    ("Power > budget?", {"color": "#f39c12"}),
    ("PM >= PM_target?", {"color": "#f39c12"}),
    ("adjust_bias", {"color": "#e74c3c"}),
    ("widen_transistors", {"color": "#e74c3c"}),
    ("reduce_current", {"color": "#e74c3c"}),
    ("generate_layout", {"color": "#2ecc71"}),
    ("FAIL", {"color": "#7f8c8d"}),
]
for n, attr in nodes_route:
    G_route.add_node(n, **attr)

edges_route = [
    ("evaluate_ppa", "PM < 45?", "check"),
    ("PM < 45?", "adjust_bias", "YES"),
    ("PM < 45?", "Gain < target?", "NO"),
    ("Gain < target?", "widen_transistors", "YES"),
    ("Gain < target?", "Power > budget?", "NO"),
    ("Power > budget?", "reduce_current", "YES"),
    ("Power > budget?", "PM >= PM_target?", "NO"),
    ("PM >= PM_target?", "generate_layout", "YES"),
    ("PM >= PM_target?", "adjust_bias", "NO"),
]
for u, v, lbl in edges_route:
    G_route.add_edge(u, v, label=lbl)

pos_route = {
    "evaluate_ppa": (0, 4),
    "PM < 45?": (0, 3),
    "adjust_bias": (-3, 2),
    "Gain < target?": (0, 2),
    "widen_transistors": (-3, 1),
    "Power > budget?": (0, 1),
    "reduce_current": (-3, 0),
    "PM >= PM_target?": (0, 0),
    "generate_layout": (3, 0),
    "FAIL": (3, 3),
}

node_colors_route = [G_route.nodes[n].get("color", "#3498db") for n in G_route.nodes()]
nx.draw_networkx_nodes(G_route, pos_route, ax=ax, node_size=2800,
                       node_color=node_colors_route, edgecolors="black", linewidths=1.5)
nx.draw_networkx_labels(G_route, pos_route, ax=ax, font_size=8, font_weight="bold")
nx.draw_networkx_edges(G_route, pos_route, ax=ax, edge_color="#2c3e50",
                       width=1.5, arrows=True, arrowsize=15, arrowstyle="-|>",
                       connectionstyle="arc3,rad=0.1")
edge_labels = {(u, v): d["label"] for u, v, d in G_route.edges(data=True)}
nx.draw_networkx_edge_labels(G_route, pos_route, edge_labels, ax=ax, font_size=7,
                             font_color="#e74c3c")
ax.set_title("PPA-Based Conditional Routing Decision Tree", fontsize=13, fontweight="bold")
ax.axis("off")
plt.tight_layout()
plt.show()
