# 3.5  Execute the EDA graph
# ---------------------------------------------------------------------------

random.seed(42)

initial_state: EDAState = {
    "spec": "Design a folded-cascode OTA with >=60 dB gain, >=100 MHz UGF, "
            ">=60 deg phase margin, <=1 mW power in 180 nm CMOS at 1.8 V supply.",
    "max_iterations": 10,
}

final_state = eda_graph.invoke(initial_state)

print(f"Status     : {final_state['status']}")
print(f"Iterations : {final_state['iteration']}")
print(f"Topology   : {final_state['topology']}")
print(f"\nPPA Achieved:")
for k, v in final_state.get("ppa_metrics", {}).get("achieved", {}).items():
    print(f"  {k}: {v}")
print(f"\nTargets Met:")
for k, v in final_state.get("ppa_metrics", {}).get("met", {}).items():
    status_mark = "PASS" if v else "FAIL"
    print(f"  {k}: {status_mark}")
