from langgraph.checkpoint.sqlite import SqliteSaver
import sqlite3

db_path = "/tmp/eda_checkpoints.db"
sqlite_conn = sqlite3.connect(db_path, check_same_thread=False)

with SqliteSaver(sqlite_conn) as sqlite_checkpointer:
    eda_graph_sqlite = eda_builder.compile(checkpointer=sqlite_checkpointer)

    config_sqlite = {"configurable": {"thread_id": "ota-prod-run-042"}}
    random.seed(99)
    final_prod = eda_graph_sqlite.invoke(initial_state, config=config_sqlite)

    print(f"Production run completed - status: {final_prod['status']}")
    print(f"Iterations: {final_prod['iteration']}")

    prod_checkpoints = list(sqlite_checkpointer.list(config_sqlite))
    print(f"Checkpoints persisted to {db_path}: {len(prod_checkpoints)}")

print("\n--- Simulating machine restart ---")
print("Re-opening the same SQLite database and reading the last checkpoint...\n")

sqlite_conn2 = sqlite3.connect(db_path, check_same_thread=False)
with SqliteSaver(sqlite_conn2) as sqlite_checkpointer2:
    restored_cp = sqlite_checkpointer2.get(config_sqlite)
    if restored_cp:
        restored_state = restored_cp["channel_values"]
        print(f"Restored state - status: {restored_state.get('status')}, "
              f"iteration: {restored_state.get('iteration')}")
    else:
        print("No checkpoint found.")
