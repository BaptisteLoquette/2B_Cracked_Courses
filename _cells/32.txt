# ---------------------------------------------------------------------------
# 8.3  Subgraph Composition Implementation
# ---------------------------------------------------------------------------

class AnalogFEState(TypedDict, total=False):
    spec: str
    topology: str
    gain_db: float
    status: str


class DigitalBEState(TypedDict, total=False):
    spec: str
    synthesis_done: bool
    timing_met: bool
    status: str


class TopLevelState(TypedDict, total=False):
    spec: str
    analog_result: dict
    digital_result: dict
    integration_status: str


# --- Analog Front-End Subgraph ---
def afe_design(state: AnalogFEState) -> dict:
    return {"topology": "folded-cascode", "gain_db": 62.5, "status": "afe_done"}

afe_builder = StateGraph(AnalogFEState)
afe_builder.add_node("afe_design", afe_design)
afe_builder.add_edge(START, "afe_design")
afe_builder.add_edge("afe_design", END)
afe_subgraph = afe_builder.compile()

# --- Digital Back-End Subgraph ---
def dbe_synthesize(state: DigitalBEState) -> dict:
    return {"synthesis_done": True, "timing_met": True, "status": "dbe_done"}

dbe_builder = StateGraph(DigitalBEState)
dbe_builder.add_node("dbe_synthesize", dbe_synthesize)
dbe_builder.add_edge(START, "dbe_synthesize")
dbe_builder.add_edge("dbe_synthesize", END)
dbe_subgraph = dbe_builder.compile()


# --- Top-Level Composition ---
def run_analog(state: TopLevelState) -> dict:
    result = afe_subgraph.invoke({"spec": state.get("spec", "")})
    return {"analog_result": result}

def run_digital(state: TopLevelState) -> dict:
    result = dbe_subgraph.invoke({"spec": state.get("spec", "")})
    return {"digital_result": result}

def integrate(state: TopLevelState) -> dict:
    afe_ok = state.get("analog_result", {}).get("status") == "afe_done"
    dbe_ok = state.get("digital_result", {}).get("status") == "dbe_done"
    status = "integration_pass" if (afe_ok and dbe_ok) else "integration_fail"
    return {"integration_status": status}


top_builder = StateGraph(TopLevelState)
top_builder.add_node("analog_fe", run_analog)
top_builder.add_node("digital_be", run_digital)
top_builder.add_node("integrate", integrate)

top_builder.add_edge(START, "analog_fe")
top_builder.add_edge(START, "digital_be")
top_builder.add_edge("analog_fe", "integrate")
top_builder.add_edge("digital_be", "integrate")
top_builder.add_edge("integrate", END)

top_graph = top_builder.compile()
top_result = top_graph.invoke({"spec": "Mixed-signal ADC SoC"})

print("Top-Level Subgraph Composition Result:")
print(f"  Analog FE  : {top_result.get('analog_result', {})}")
print(f"  Digital BE : {top_result.get('digital_result', {})}")
print(f"  Integration: {top_result.get('integration_status')}")

# Visualize top-level graph
draw_langgraph(top_graph, "SoC Design - Subgraph Composition")
