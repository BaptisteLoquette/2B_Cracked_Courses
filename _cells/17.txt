def route_after_ppa(state: EDAState) -> str:
    """Conditional router based on PPA evaluation status."""
    status = state.get("status", "running")
    if status == "converged":
        return "generate_layout"
    elif status == "failed":
        return "__end__"
    else:
        return "generate_netlist"


eda_builder = StateGraph(EDAState)

eda_builder.add_node("parse_spec", parse_spec)
eda_builder.add_node("select_topology", select_topology)
eda_builder.add_node("generate_netlist", generate_netlist)
eda_builder.add_node("run_simulation", run_simulation)
eda_builder.add_node("evaluate_ppa", evaluate_ppa)
eda_builder.add_node("generate_layout", generate_layout)

eda_builder.add_edge(START, "parse_spec")
eda_builder.add_edge("parse_spec", "select_topology")
eda_builder.add_edge("select_topology", "generate_netlist")
eda_builder.add_edge("generate_netlist", "run_simulation")
eda_builder.add_edge("run_simulation", "evaluate_ppa")
eda_builder.add_conditional_edges(
    "evaluate_ppa",
    route_after_ppa,
    {
        "generate_netlist": "generate_netlist",
        "generate_layout": "generate_layout",
        "__end__": END,
    },
)
eda_builder.add_edge("generate_layout", END)

eda_graph = eda_builder.compile()
print("EDA graph compiled successfully.")
