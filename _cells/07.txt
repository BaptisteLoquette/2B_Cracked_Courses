def node_parse(state: SimpleState) -> dict:
    """Parse and normalize the input text."""
    raw = state.get("input_text", "")
    return {"processed": raw.strip().upper()}


def node_analyze(state: SimpleState) -> dict:
    """Analyze the processed text: compute length and word count."""
    text = state.get("processed", "")
    analysis = f"length={len(text)}, words={len(text.split())}"
    return {"result": analysis}


def node_report(state: SimpleState) -> dict:
    """Generate a final report string."""
    return {"result": f"Report >> {state.get('result', '')}"}


print("Node functions defined: node_parse, node_analyze, node_report")
