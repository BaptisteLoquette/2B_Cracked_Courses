# 9.3  Map-Reduce: Parallel Corner Simulations
# ---------------------------------------------------------------------------

class CornerSimState(TypedDict, total=False):
    netlist: str
    corners: list[str]
    corner_results: dict
    worst_case: dict
    status: str


def prepare_corners(state: CornerSimState) -> dict:
    return {
        "corners": ["TT", "FF", "SS", "SF", "FS"],
        "status": "corners_prepared",
    }


def simulate_all_corners(state: CornerSimState) -> dict:
    """Simulate all corners (in production these would be parallel jobs)."""
    results = {}
    for corner in state.get("corners", []):
        base_gain = 60.0
        corner_offsets = {"TT": 0, "FF": 3.5, "SS": -4.2, "SF": -1.8, "FS": 1.2}
        gain = base_gain + corner_offsets.get(corner, 0) + random.uniform(-0.5, 0.5)
        pm = 62 + random.uniform(-8, 5)
        power = 0.85 + random.uniform(-0.1, 0.2)
        results[corner] = {
            "gain_db": round(gain, 2),
            "phase_margin_deg": round(pm, 1),
            "power_mw": round(power, 4),
        }
    return {"corner_results": results}


def find_worst_case(state: CornerSimState) -> dict:
    """Reduce: find worst-case across all corners."""
    results = state.get("corner_results", {})
    if not results:
        return {"status": "no_results"}

    worst_gain_corner = min(results, key=lambda c: results[c]["gain_db"])
    worst_pm_corner = min(results, key=lambda c: results[c]["phase_margin_deg"])
    worst_power_corner = max(results, key=lambda c: results[c]["power_mw"])

    worst = {
        "gain": {"corner": worst_gain_corner, **results[worst_gain_corner]},
        "phase_margin": {"corner": worst_pm_corner, **results[worst_pm_corner]},
        "power": {"corner": worst_power_corner, **results[worst_power_corner]},
    }
    return {"worst_case": worst, "status": "worst_case_found"}


corner_builder = StateGraph(CornerSimState)
corner_builder.add_node("prepare_corners", prepare_corners)
corner_builder.add_node("simulate_all_corners", simulate_all_corners)
corner_builder.add_node("find_worst_case", find_worst_case)

corner_builder.add_edge(START, "prepare_corners")
corner_builder.add_edge("prepare_corners", "simulate_all_corners")
corner_builder.add_edge("simulate_all_corners", "find_worst_case")
corner_builder.add_edge("find_worst_case", END)

corner_graph = corner_builder.compile()

random.seed(42)
corner_result = corner_graph.invoke({"netlist": "* folded-cascode netlist\n.end"})

print("Corner Simulation Results (Map-Reduce):")
print(f"{'Corner':>6} | {'Gain(dB)':>9} | {'PM(deg)':>8} | {'Power(mW)':>10}")
print("-" * 42)
for corner, metrics in corner_result["corner_results"].items():
    print(f"{corner:>6} | {metrics['gain_db']:>9.2f} | {metrics['phase_margin_deg']:>8.1f} | "
          f"{metrics['power_mw']:>10.4f}")

print(f"\nWorst-Case Analysis:")
for metric, info in corner_result["worst_case"].items():
    val_key = {"gain": "gain_db", "phase_margin": "phase_margin_deg", "power": "power_mw"}[metric]
    print(f"  {metric}: corner={info['corner']}, value={info[val_key]}")
